# Plan 21-02: Signal Routing Configuration

## Objective

Build a flexible signal routing system that allows users to route TradingView signals to specific accounts, all accounts, or based on rules. Implements the core routing logic referenced by ACCT-03.

## Context

**Phase Requirements:**
- ACCT-03: Route signals to specific accounts or all accounts

**Existing Infrastructure:**
- `WebhookConfig` model has `routing_rules` JSON column and `default_account_id`
- `SignalService._get_target_accounts()` returns accounts for signal execution
- `Signal` entity has `target_accounts: List[AccountId]`
- `signal_processor.py` has basic broker/account detection from payload

**Key Changes Needed:**
1. Define routing rule schema and evaluation engine
2. Integrate routing with webhook processing
3. Support "route to all", "route to specific", and "conditional routing"

### Task 1: Define Routing Rule Schema and Engine

```python
# app/domain/services/routing_service.py
from dataclasses import dataclass
from typing import List, Optional, Any, Dict
from enum import Enum

class RoutingStrategy(str, Enum):
    ALL_ACCOUNTS = "all_accounts"           # Route to all active accounts
    SPECIFIC_ACCOUNTS = "specific_accounts" # Route to listed account IDs
    RULES_BASED = "rules_based"            # Evaluate routing rules
    DEFAULT_ONLY = "default_only"          # Only default account

class RoutingOperator(str, Enum):
    EQUALS = "eq"
    NOT_EQUALS = "neq"
    CONTAINS = "contains"
    STARTS_WITH = "starts_with"
    IN_LIST = "in"
    GREATER_THAN = "gt"
    LESS_THAN = "lt"

@dataclass
class RoutingRule:
    """A single routing rule"""
    field: str              # "symbol", "action", "source", "strategy_name"
    operator: RoutingOperator
    value: Any              # Comparison value
    target_account_ids: List[int]
    priority: int = 0

@dataclass
class RoutingConfig:
    """Complete routing configuration"""
    strategy: RoutingStrategy
    default_account_id: Optional[int] = None
    specific_account_ids: List[int] = None
    rules: List[RoutingRule] = None

class RoutingEngine:
    """Evaluates routing rules against signal data"""

    def __init__(self, config: RoutingConfig, available_accounts: List[int]):
        self.config = config
        self.available_accounts = set(available_accounts)

    def resolve_target_accounts(self, signal_data: Dict[str, Any]) -> List[int]:
        """Resolve which accounts should receive the signal"""
        if self.config.strategy == RoutingStrategy.ALL_ACCOUNTS:
            return list(self.available_accounts)

        elif self.config.strategy == RoutingStrategy.SPECIFIC_ACCOUNTS:
            return [aid for aid in self.config.specific_account_ids
                    if aid in self.available_accounts]

        elif self.config.strategy == RoutingStrategy.DEFAULT_ONLY:
            if self.config.default_account_id in self.available_accounts:
                return [self.config.default_account_id]
            return []

        elif self.config.strategy == RoutingStrategy.RULES_BASED:
            return self._evaluate_rules(signal_data)

        return []

    def _evaluate_rules(self, signal_data: Dict[str, Any]) -> List[int]:
        """Evaluate rules and return matching account IDs"""
        matched_accounts = set()

        # Sort rules by priority (higher first)
        sorted_rules = sorted(self.config.rules or [], key=lambda r: -r.priority)

        for rule in sorted_rules:
            if self._rule_matches(rule, signal_data):
                for account_id in rule.target_account_ids:
                    if account_id in self.available_accounts:
                        matched_accounts.add(account_id)

        # Fallback to default if no rules matched
        if not matched_accounts and self.config.default_account_id:
            if self.config.default_account_id in self.available_accounts:
                matched_accounts.add(self.config.default_account_id)

        return list(matched_accounts)

    def _rule_matches(self, rule: RoutingRule, signal_data: Dict[str, Any]) -> bool:
        """Check if a single rule matches the signal"""
        field_value = signal_data.get(rule.field)
        if field_value is None:
            return False

        op = rule.operator
        compare_value = rule.value

        if op == RoutingOperator.EQUALS:
            return str(field_value).lower() == str(compare_value).lower()
        elif op == RoutingOperator.NOT_EQUALS:
            return str(field_value).lower() != str(compare_value).lower()
        elif op == RoutingOperator.CONTAINS:
            return str(compare_value).lower() in str(field_value).lower()
        elif op == RoutingOperator.STARTS_WITH:
            return str(field_value).lower().startswith(str(compare_value).lower())
        elif op == RoutingOperator.IN_LIST:
            return str(field_value).lower() in [str(v).lower() for v in compare_value]
        elif op == RoutingOperator.GREATER_THAN:
            return float(field_value) > float(compare_value)
        elif op == RoutingOperator.LESS_THAN:
            return float(field_value) < float(compare_value)

        return False
```

**Acceptance:**
- RoutingEngine evaluates all strategies correctly
- Rules can match on symbol, action, source fields
- Priority ordering works
- Default fallback works

```python
from app.domain.services.routing_service import RoutingEngine, RoutingConfig, RoutingStrategy, RoutingRule

async def process_webhook_signal(
    webhook_config: WebhookConfig,
    payload: dict,
    user_id: int,
    db: Session
):
    """Process incoming webhook with routing"""
    # Get user's active, signal-enabled accounts
    accounts = db.query(TradingAccount).filter(
        TradingAccount.user_id == user_id,
        TradingAccount.is_active == True,
        TradingAccount.is_signal_enabled == True
    ).all()

    available_account_ids = [a.id for a in accounts]

    # Build routing config from webhook config
    routing_config = build_routing_config(webhook_config)

    # Create routing engine
    engine = RoutingEngine(routing_config, available_account_ids)

    # Build signal data for rule evaluation
    signal_data = {
        "symbol": payload.get("ticker") or payload.get("symbol"),
        "action": payload.get("action") or payload.get("signal"),
        "source": webhook_config.source,
        "strategy_name": payload.get("strategy_name"),
        "strategy_id": payload.get("strategy_id"),
    }

    # Resolve target accounts
    target_account_ids = engine.resolve_target_accounts(signal_data)

    if not target_account_ids:
        logger.warning(f"No target accounts resolved for signal {signal_data}")
        return {"success": False, "error": "No target accounts matched routing rules"}

    # Execute signal on each target account
    results = []
    for account_id in target_account_ids:
        result = await execute_signal_on_account(payload, account_id, db)
        results.append(result)

    return {
        "success": any(r.get("success") for r in results),
        "accounts_targeted": len(target_account_ids),
        "results": results
    }

def build_routing_config(webhook_config: WebhookConfig) -> RoutingConfig:
    """Convert WebhookConfig to RoutingConfig"""
    # Determine strategy from config
    if webhook_config.routing_rules:
        strategy = RoutingStrategy.RULES_BASED
        rules = [
            RoutingRule(
                field=r.get("condition", {}).get("field", "symbol"),
                operator=RoutingOperator(r.get("condition", {}).get("operator", "eq")),
                value=r.get("condition", {}).get("value"),
                target_account_ids=[r.get("target_account_id")],
                priority=r.get("priority", 0)
            )
            for r in webhook_config.routing_rules
        ]
    else:
        strategy = RoutingStrategy.DEFAULT_ONLY
        rules = []

    return RoutingConfig(
        strategy=strategy,
        default_account_id=webhook_config.default_account_id,
        rules=rules
    )
```

**Acceptance:**
- Webhooks use routing engine
- Multiple accounts can receive same signal
- Rules are evaluated correctly
- Default fallback works

```python
async def _get_target_accounts(self, signal: Signal) -> List[Account]:
    """Get accounts to execute signal on, respecting settings"""
    accounts = []

    if signal.target_accounts:
        # Specific accounts targeted
        for account_id in signal.target_accounts:
            account = await self._account_repo.get_by_id(account_id)
            if self._account_can_receive_signal(account):
                accounts.append(account)
    else:
        # All signal-enabled connected accounts
        all_accounts = await self._account_repo.get_connected()
        accounts = [a for a in all_accounts if self._account_can_receive_signal(a)]

    # Sort by signal_priority (higher first)
    accounts.sort(key=lambda a: getattr(a, 'signal_priority', 0), reverse=True)

    return accounts

def _account_can_receive_signal(self, account: Optional[Account]) -> bool:
    """Check if account can receive signals"""
    if not account:
        return False
    if not account.is_active:
        return False
    if not account.is_connected:
        return False
    if not getattr(account, 'is_signal_enabled', True):
        return False
    return True
```

**Acceptance:**
- Signal execution respects is_signal_enabled
- Account priority ordering works
- Inactive/disconnected accounts excluded

```python
@router.put("/{config_id}/routing")
async def update_routing(
    config_id: int,
    routing: RoutingUpdate,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Update webhook routing configuration"""
    config = get_webhook_config_or_404(config_id, current_user.id, db)

    # Validate specific account IDs belong to user
    if routing.specific_account_ids:
        for aid in routing.specific_account_ids:
            account = db.query(TradingAccount).filter(
                TradingAccount.id == aid,
                TradingAccount.user_id == current_user.id
            ).first()
            if not account:
                raise HTTPException(400, f"Account {aid} not found")

    # Update routing
    config.routing_strategy = routing.strategy
    config.default_account_id = routing.default_account_id
    config.specific_account_ids = routing.specific_account_ids
    config.routing_rules = [r.dict() for r in routing.rules] if routing.rules else []

    db.commit()
    return {"message": "Routing updated", "strategy": config.routing_strategy}

class RoutingUpdate(BaseModel):
    strategy: str  # all_accounts, specific_accounts, rules_based, default_only
    default_account_id: Optional[int] = None
    specific_account_ids: Optional[List[int]] = None
    rules: Optional[List[RoutingRuleCreate]] = None
```

**Acceptance:**
- Routing strategy can be updated
- Account IDs validated against user ownership
- Rules can be added/updated
