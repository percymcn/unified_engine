# Plan 16-01: Tradovate OAuth Backend Endpoints

## Objective
Create FastAPI endpoints to initiate Tradovate OAuth flow and handle the callback with code exchange.

## Context
- **Phase**: 16 - Tradovate OAuth
- **Requirements**: CONN-03 (OAuth redirect flow), CONN-04 (callback handler)
- **Current State**: No OAuth endpoints exist; uses username/password auth
- **Target State**: `/api/v1/auth/tradovate/authorize` and `/api/v1/auth/tradovate/callback`

## Success Criteria
1. Initiate endpoint generates correct Tradovate authorization URL with state
2. Callback endpoint exchanges authorization code for access/refresh tokens
3. State parameter validates to prevent CSRF attacks
4. Tokens returned to frontend for secure storage

## Research Summary

### Tradovate OAuth Flow
1. **Authorization URL**: `https://trader.tradovate.com/oauth`
   - Params: `client_id`, `redirect_uri`, `response_type=code`, `state`
2. **Token Exchange**: `https://live.tradovateapi.com/v1/auth/oauthtoken`
   - POST with: `grant_type=authorization_code`, `code`, `client_id`, `client_secret`, `redirect_uri`
3. **Token Response**: `access_token`, `refresh_token`, `expires_in`, `token_type`

### Environment Selection
- Demo: `https://demo.tradovateapi.com/v1`
- Live: `https://live.tradovateapi.com/v1`

### Task 2: Create Tradovate OAuth Router
**File**: `app/routers/tradovate_oauth.py` (NEW)
**Implementation**:

```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
import secrets
import httpx
from urllib.parse import urlencode

from app.db.database import get_db
from app.routers.auth import get_current_user
from app.models.models import User
from app.core.config import settings
from app.core.encryption import encrypt_dict

router = APIRouter(prefix="/api/v1/auth/tradovate", tags=["tradovate-oauth"])

# In-memory state store (use Redis in production)
_oauth_states: dict[str, dict] = {}

@router.get("/authorize")
async def initiate_oauth(
    environment: str = Query("demo", regex="^(demo|live)$"),
    current_user: User = Depends(get_current_user),
):
    """Generate Tradovate OAuth authorization URL"""
    if not settings.TRADOVATE_CLIENT_ID:
        raise HTTPException(400, "Tradovate OAuth not configured")

    # Generate state for CSRF protection
    state = secrets.token_urlsafe(32)
    _oauth_states[state] = {
        "user_id": current_user.id,
        "environment": environment,
    }

    params = {
        "client_id": settings.TRADOVATE_CLIENT_ID,
        "redirect_uri": settings.TRADOVATE_OAUTH_REDIRECT_URI,
        "response_type": "code",
        "state": state,
    }

    auth_url = f"https://trader.tradovate.com/oauth?{urlencode(params)}"
    return {"authorization_url": auth_url, "state": state}

@router.get("/callback")
async def oauth_callback(
    code: str = Query(...),
    state: str = Query(...),
    db: Session = Depends(get_db),
):
    """Handle OAuth callback and exchange code for tokens"""
    # Validate state
    state_data = _oauth_states.pop(state, None)
    if not state_data:
        raise HTTPException(400, "Invalid or expired OAuth state")

    environment = state_data["environment"]
    user_id = state_data["user_id"]

    # Select API base URL
    api_base = (
        "https://live.tradovateapi.com/v1"
        if environment == "live"
        else "https://demo.tradovateapi.com/v1"
    )

    # Exchange code for tokens
    async with httpx.AsyncClient() as client:
        response = await client.post(
            f"{api_base}/auth/oauthtoken",
            data={
                "grant_type": "authorization_code",
                "code": code,
                "client_id": settings.TRADOVATE_CLIENT_ID,
                "client_secret": settings.TRADOVATE_CLIENT_SECRET,
                "redirect_uri": settings.TRADOVATE_OAUTH_REDIRECT_URI,
            },
        )

        if response.status_code != 200:
            raise HTTPException(400, f"Token exchange failed: {response.text}")

        tokens = response.json()

    # Return tokens to frontend (frontend will store them)
    return {
        "access_token": tokens["accessToken"],
        "refresh_token": tokens.get("refreshToken"),
        "expires_in": tokens.get("expiresIn", 3600),
        "environment": environment,
        "user_id": user_id,
    }
```

### Task 4: Add Next.js BFF Route for Callback
**File**: `ui-next/src/app/api/auth/tradovate/callback/route.ts` (NEW)
**Purpose**: Handle the browser redirect from Tradovate OAuth

```typescript
import { NextResponse } from 'next/server';

const BACKEND_URL = process.env.BACKEND_URL || 'http://localhost:8765';

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const code = searchParams.get('code');
  const state = searchParams.get('state');
  const error = searchParams.get('error');

  if (error) {
    return NextResponse.redirect(
      new URL(`/dashboard/settings/accounts?error=${error}`, request.url)
    );
  }

  if (!code || !state) {
    return NextResponse.redirect(
      new URL('/dashboard/settings/accounts?error=missing_params', request.url)
    );
  }

  try {
    // Exchange code via backend
    const response = await fetch(
      `${BACKEND_URL}/api/v1/auth/tradovate/callback?code=${code}&state=${state}`
    );

    if (!response.ok) {
      throw new Error('Token exchange failed');
    }

    const tokens = await response.json();

    // Redirect with tokens in URL fragment (secure)
    // Frontend JavaScript will read and store them
    return NextResponse.redirect(
      new URL(
        `/dashboard/settings/accounts?tradovate_connected=true#tokens=${encodeURIComponent(
          JSON.stringify(tokens)
        )}`,
        request.url
      )
    );
  } catch (error) {
    return NextResponse.redirect(
      new URL('/dashboard/settings/accounts?error=token_exchange_failed', request.url)
    );
  }
}
```
