    ```python
    class TradeLockerAdapter(BrokerPort):
        """
        TradeLocker broker adapter implementing domain BrokerPort.

        Wraps TradeLockerExecutor and converts between domain value objects
        and executor primitives.
        """

        def __init__(self, executor: TradeLockerExecutor = None):
            self._executor = executor

        @property
        def broker_type(self) -> BrokerType:
            return BrokerType.TRADELOCKER

        @property
        def is_connected(self) -> bool:
            return self._executor is not None and self._executor.is_connected

        async def connect(self, credentials: Dict[str, Any]) -> None:
            """Connect to TradeLocker API."""
            if self._executor is None:
                self._executor = TradeLockerExecutor()
            await self._executor.connect(
                email=credentials.get("email"),
                password=credentials.get("password"),
                server=credentials.get("server"),
            )

        async def disconnect(self) -> None:
            """Disconnect from TradeLocker."""
            if self._executor:
                await self._executor.disconnect()

        async def authenticate(self, account_id: AccountId) -> bool:
            """Authenticate with account."""
            # TradeLocker authenticates on connect
            return self.is_connected

        async def place_order(
            self,
            account_id: AccountId,
            symbol: Symbol,
            side: OrderSide,
            order_type: OrderType,
            volume: Volume,
            price: Optional[Price] = None,
            stop_loss: Optional[Price] = None,
            take_profit: Optional[Price] = None,
        ) -> Order:
            """Place order via TradeLocker."""
            # Convert domain types to executor primitives
            result = await self._executor.place_order(
                symbol=symbol.value,
                side=side.value,
                order_type=order_type.value,
                volume=float(volume.value),
                price=float(price.value) if price else None,
                stop_loss=float(stop_loss.value) if stop_loss else None,
                take_profit=float(take_profit.value) if take_profit else None,
            )
            # Convert executor result to domain Order
            return self._to_domain_order(result, account_id)

        async def close_position(
            self,
            account_id: AccountId,
            position_id: PositionId,
            volume: Optional[Volume] = None,
        ) -> Trade:
            """Close position via TradeLocker."""
            result = await self._executor.close_position(
                position_id=str(position_id),
                volume=float(volume.value) if volume else None,
            )
            return self._to_domain_trade(result, account_id)

        async def modify_position(
            self,
            account_id: AccountId,
            position_id: PositionId,
            stop_loss: Optional[Price] = None,
            take_profit: Optional[Price] = None,
        ) -> Position:
            """Modify position SL/TP."""
            result = await self._executor.modify_position(
                position_id=str(position_id),
                stop_loss=float(stop_loss.value) if stop_loss else None,
                take_profit=float(take_profit.value) if take_profit else None,
            )
            return self._to_domain_position(result, account_id)

        async def get_positions(self, account_id: AccountId) -> List[Position]:
            """Get open positions."""
            results = await self._executor.get_positions()
            return [self._to_domain_position(p, account_id) for p in results]

        async def get_orders(self, account_id: AccountId) -> List[Order]:
            """Get pending orders."""
            results = await self._executor.get_orders()
            return [self._to_domain_order(o, account_id) for o in results]

        async def get_account_info(self, account_id: AccountId) -> Dict[str, Any]:
            """Get account balance and details."""
            return await self._executor.get_account_info()

        async def cancel_order(self, account_id: AccountId, order_id: OrderId) -> bool:
            """Cancel pending order."""
            return await self._executor.cancel_order(str(order_id))

        # Private conversion methods
        def _to_domain_order(self, data: Dict, account_id: AccountId) -> Order: ...
        def _to_domain_position(self, data: Dict, account_id: AccountId) -> Position: ...
        def _to_domain_trade(self, data: Dict, account_id: AccountId) -> Trade: ...
    ```
