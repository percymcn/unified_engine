# Plan 22-01: Risk Enforcement Service

## Objective

Build the core risk enforcement engine that evaluates and enforces trade/signal limits before execution. This creates the foundation for all risk management features.

## Context

**Phase Requirements:**
- RISK-01: Maximum signals per day (configurable limit)
- RISK-02: Maximum concurrent trades per broker
- RISK-03: Maximum concurrent trades total (across all brokers)
- RISK-04: Block signals when any limit reached (with notification)
- RISK-11: Per-symbol trade limits (max positions per instrument)
- RISK-12: Cooldown period between trades (configurable delay)

**Existing Infrastructure:**
- `TradingAccount` has risk limit columns: `max_daily_trades`, `max_open_positions`, `trade_cooldown_seconds`
- `signal_processor.py` has basic `_check_risk_limits()` that only checks global `MAX_POSITION_SIZE`
- `SignalService` processes signals but doesn't enforce account-specific limits
- Global `settings.RISK_MANAGEMENT_ENABLED` flag exists but isn't per-user

**Key Changes Needed:**
1. Create `RiskEnforcementService` domain service
2. Add `RejectedSignal` model for logging blocked signals
3. Track daily signal/trade counts per account
4. Implement cooldown tracking per account/symbol
5. Integrate with signal processing pipeline

# app/domain/services/daily_counter_service.py
from datetime import date, datetime
from typing import Dict, Optional
from dataclasses import dataclass

@dataclass
class DailyCounters:
    """Daily counters for an account"""
    account_id: int
    date: date
    signals_received: int = 0
    trades_executed: int = 0
    trades_by_symbol: Dict[str, int] = None
    last_trade_at: Optional[datetime] = None

    def __post_init__(self):
        if self.trades_by_symbol is None:
            self.trades_by_symbol = {}

class DailyCounterService:
    """Tracks daily signal and trade counts per account"""

    def __init__(self, repository):
        self._repo = repository

    async def get_counters(self, account_id: int, for_date: date = None) -> DailyCounters:
        """Get or create daily counters for account"""
        if for_date is None:
            for_date = date.today()
        return await self._repo.get_or_create(account_id, for_date)

    async def increment_signals(self, account_id: int) -> DailyCounters:
        """Increment daily signal count"""
        counters = await self.get_counters(account_id)
        counters.signals_received += 1
        await self._repo.save(counters)
        return counters

    async def increment_trades(self, account_id: int, symbol: str) -> DailyCounters:
        """Increment daily trade count and per-symbol count"""
        counters = await self.get_counters(account_id)
        counters.trades_executed += 1
        counters.trades_by_symbol[symbol] = counters.trades_by_symbol.get(symbol, 0) + 1
        counters.last_trade_at = datetime.utcnow()
        await self._repo.save(counters)
        return counters

    async def get_symbol_trade_count(self, account_id: int, symbol: str) -> int:
        """Get today's trade count for specific symbol"""
        counters = await self.get_counters(account_id)
        return counters.trades_by_symbol.get(symbol, 0)
```

**Acceptance:**
- Can track daily signal counts per account
- Can track daily trade counts per account
- Can track per-symbol trade counts
- Counters reset daily (date-based)

# app/domain/services/risk_enforcement_service.py
from dataclasses import dataclass
from typing import Optional, List, Tuple
from datetime import datetime, timedelta
from enum import Enum

class RiskCheckResult(Enum):
    PASSED = "passed"
    BLOCKED = "blocked"

@dataclass
class RiskViolation:
    """Details of a risk limit violation"""
    reason: str  # RejectedSignalReason value
    detail: str
    limit_value: float
    current_value: float

@dataclass
class RiskEvaluation:
    """Result of risk evaluation"""
    result: RiskCheckResult
    violations: List[RiskViolation]

    @property
    def passed(self) -> bool:
        return self.result == RiskCheckResult.PASSED

    @property
    def first_violation(self) -> Optional[RiskViolation]:
        return self.violations[0] if self.violations else None

class RiskEnforcementService:
    """
    Evaluates and enforces risk limits before signal execution.

    Checks performed:
    1. Daily signal limit (max_daily_trades)
    2. Concurrent position limit (max_open_positions)
    3. Per-symbol position limit
    4. Trade cooldown
    """

    def __init__(self, counter_service, position_service):
        self._counters = counter_service
        self._positions = position_service

    async def evaluate(
        self,
        account_id: int,
        symbol: str,
        action: str,
        account_settings: dict
    ) -> RiskEvaluation:
        """
        Evaluate all risk limits for a signal.

        Args:
            account_id: Target account ID
            symbol: Trading symbol
            action: Signal action (buy, sell, close)
            account_settings: Account risk settings dict

        Returns:
            RiskEvaluation with pass/block result and any violations
        """
        violations = []

        # Skip checks for close actions
        if action.lower() in ('close', 'close_all'):
            return RiskEvaluation(RiskCheckResult.PASSED, [])

        # Check 1: Daily trade limit
        if account_settings.get('max_daily_trades'):
            counters = await self._counters.get_counters(account_id)
            if counters.trades_executed >= account_settings['max_daily_trades']:
                violations.append(RiskViolation(
                    reason="daily_limit",
                    detail=f"Daily trade limit reached ({counters.trades_executed}/{account_settings['max_daily_trades']})",
                    limit_value=account_settings['max_daily_trades'],
                    current_value=counters.trades_executed
                ))

        # Check 2: Concurrent position limit (total)
        if account_settings.get('max_open_positions'):
            open_positions = await self._positions.count_open(account_id)
            if open_positions >= account_settings['max_open_positions']:
                violations.append(RiskViolation(
                    reason="concurrent_limit",
                    detail=f"Max open positions reached ({open_positions}/{account_settings['max_open_positions']})",
                    limit_value=account_settings['max_open_positions'],
                    current_value=open_positions
                ))

        # Check 3: Per-symbol position limit
        max_per_symbol = account_settings.get('max_positions_per_symbol', 1)
        symbol_positions = await self._positions.count_open_for_symbol(account_id, symbol)
        if symbol_positions >= max_per_symbol:
            violations.append(RiskViolation(
                reason="symbol_limit",
                detail=f"Max positions for {symbol} reached ({symbol_positions}/{max_per_symbol})",
                limit_value=max_per_symbol,
                current_value=symbol_positions
            ))

        # Check 4: Trade cooldown
        if account_settings.get('trade_cooldown_seconds'):
            counters = await self._counters.get_counters(account_id)
            if counters.last_trade_at:
                elapsed = (datetime.utcnow() - counters.last_trade_at).total_seconds()
                if elapsed < account_settings['trade_cooldown_seconds']:
                    remaining = account_settings['trade_cooldown_seconds'] - elapsed
                    violations.append(RiskViolation(
                        reason="cooldown",
                        detail=f"Trade cooldown active ({remaining:.0f}s remaining)",
                        limit_value=account_settings['trade_cooldown_seconds'],
                        current_value=elapsed
                    ))

        result = RiskCheckResult.BLOCKED if violations else RiskCheckResult.PASSED
        return RiskEvaluation(result, violations)

    async def log_rejection(
        self,
        user_id: int,
        account_id: int,
        symbol: str,
        action: str,
        violation: RiskViolation,
        source: str = None,
        webhook_config_id: int = None,
        original_payload: dict = None
    ):
        """Log a rejected signal to the database"""
        # This will be implemented to insert into RejectedSignal table
        pass
```

**Acceptance:**
- Evaluates daily trade limits
- Evaluates concurrent position limits
- Evaluates per-symbol limits
- Evaluates trade cooldown
- Returns structured evaluation result

async def process_routed_signal(
    webhook_config: WebhookConfig,
    payload: dict,
    target_account_ids: List[int],
    db: Session
):
    """Process signal with risk enforcement"""
    from app.domain.services.risk_enforcement_service import RiskEnforcementService

    results = []
    symbol = payload.get("ticker") or payload.get("symbol")
    action = payload.get("action") or payload.get("signal")

    for account_id in target_account_ids:
        # Get account with settings
        account = db.query(TradingAccount).filter(TradingAccount.id == account_id).first()
        if not account:
            continue

        # Build account settings dict
        account_settings = {
            'max_daily_trades': account.max_daily_trades,
            'max_open_positions': account.max_open_positions,
            'max_positions_per_symbol': getattr(account, 'max_positions_per_symbol', 1),
            'trade_cooldown_seconds': account.trade_cooldown_seconds,
        }

        # Evaluate risk
        risk_service = get_risk_enforcement_service()
        evaluation = await risk_service.evaluate(
            account_id=account_id,
            symbol=symbol,
            action=action,
            account_settings=account_settings
        )

        if not evaluation.passed:
            # Log rejection
            violation = evaluation.first_violation
            await risk_service.log_rejection(
                user_id=account.user_id,
                account_id=account_id,
                symbol=symbol,
                action=action,
                violation=violation,
                source=webhook_config.source,
                webhook_config_id=webhook_config.id,
                original_payload=payload
            )
            results.append({
                "account_id": account_id,
                "success": False,
                "blocked": True,
                "reason": violation.detail
            })
            continue

        # Execute signal
        result = await execute_signal_on_account(payload, account_id, db)

        # If successful, increment counters
        if result.get("success"):
            await get_counter_service().increment_trades(account_id, symbol)

        results.append(result)

    return results
```

**Acceptance:**
- Risk check runs before signal execution
- Blocked signals are logged with reason
- Successful trades increment counters
- Results include block reason when applicable

# app/routers/risk.py
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from typing import List
from datetime import datetime, timedelta

router = APIRouter(prefix="/api/risk", tags=["risk"])

@router.get("/rejected-signals")
async def get_rejected_signals(
    limit: int = Query(50, le=100),
    offset: int = 0,
    account_id: Optional[int] = None,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Get rejected signals for current user"""
    query = db.query(RejectedSignal).filter(
        RejectedSignal.user_id == current_user.id
    )

    if account_id:
        query = query.filter(RejectedSignal.account_id == account_id)

    total = query.count()
    signals = query.order_by(RejectedSignal.created_at.desc())\
        .offset(offset).limit(limit).all()

    return {
        "total": total,
        "signals": [
            {
                "id": s.id,
                "account_id": s.account_id,
                "symbol": s.symbol,
                "action": s.action,
                "reason": s.reason.value,
                "reason_detail": s.reason_detail,
                "created_at": s.created_at.isoformat()
            }
            for s in signals
        ]
    }

@router.get("/daily-stats/{account_id}")
async def get_daily_stats(
    account_id: int,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """Get daily risk stats for an account"""
    # Verify ownership
    account = db.query(TradingAccount).filter(
        TradingAccount.id == account_id,
        TradingAccount.user_id == current_user.id
    ).first()
    if not account:
        raise HTTPException(404, "Account not found")

    counters = await get_counter_service().get_counters(account_id)

    return {
        "account_id": account_id,
        "date": counters.date.isoformat(),
        "signals_received": counters.signals_received,
        "trades_executed": counters.trades_executed,
        "trades_by_symbol": counters.trades_by_symbol,
        "limits": {
            "max_daily_trades": account.max_daily_trades,
            "max_open_positions": account.max_open_positions,
            "trade_cooldown_seconds": account.trade_cooldown_seconds
        }
    }
```

**Acceptance:**
