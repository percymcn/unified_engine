```python
class SymbolAlias(Base):

    user_id = Column(Integer, ForeignKey("users.id"), nullable=False)
    source_symbol = Column(String(50), nullable=False)
    broker_type = Column(String(50), nullable=False)
    target_symbol = Column(String(50), nullable=False)
    is_auto_detected = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.utcnow)
    updated_at = Column(DateTime, onupdate=datetime.utcnow)
```

```python
class SymbolNormalizationService:
    # Common suffix patterns to strip for base symbol extraction
    SUFFIX_PATTERNS = [
        r'\.pro$', r'\.raw$', r'\.std$', r'\.mini$',
        r'PR$', r'RAW$', r'STD$', r'MINI$',
        r'_SB$', r'\.SB$',  # Spread betting
        r'\d{2}$',  # Futures month codes like "NQ25"
    ]

    def normalize_symbol(self, symbol: str) -> str:
        """Extract base symbol by removing common suffixes."""

    def get_variations(self, base_symbol: str) -> List[str]:
        """Generate common variations for a base symbol."""

    async def resolve_symbol(
        self,
        user_id: int,
        source_symbol: str,
        broker_type: str,
        available_symbols: List[str]
    ) -> Optional[str]:
        """
        Resolve TradingView symbol to broker format.
        1. Check user aliases first
        2. Check auto-detected aliases
        3. Try fuzzy matching against available symbols
        """
```

```python
# In signal_processor.py, after parsing webhook:
async def _resolve_symbol_for_broker(
    self,
    source_symbol: str,
    user_id: int,
    broker_type: str,
    account_id: str
) -> str:
    """Resolve TradingView symbol to broker format."""
    # Get available symbols from broker
    broker = self._get_broker(broker_type, account_id)
    available_symbols = await broker.get_symbols()

    # Use normalization service
    resolved = await self.symbol_service.resolve_symbol(
        user_id=user_id,
        source_symbol=source_symbol,
        broker_type=broker_type,
        available_symbols=available_symbols
    )

    if resolved:
        return resolved

    # Fallback to original symbol (let broker handle error)
    logger.warning(f"Could not resolve symbol {source_symbol} for {broker_type}")
    return source_symbol
```

**Acceptance:**
- Signal processor calls symbol resolution before execution
- Resolution failure logs warning but doesn't block (graceful degradation)
- Resolved symbol used in order placement
