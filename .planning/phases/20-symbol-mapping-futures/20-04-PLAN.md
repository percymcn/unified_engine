# Plan 20-04: Futures Contract Support

## Objective

Implement futures contract expiration tracking and rollover support, especially for TopStep/ProjectX brokers, with auto-roll notifications.

## Context

**Phase Requirements:**
- SYM-05: Support futures contract rollover (especially TopStep/ProjectX)
- SYM-06: Contract expiration tracking and auto-roll notifications

**Discovery Findings:**
- Tradovate executor uses contract lookups: `tradovate_executor.py:396`
- ProjectX SDK (`project-x-py`) handles futures contracts
- Futures contracts have month codes (e.g., NQH25 = NQ March 2025)
- Rollover typically happens a few days before expiration
- Common futures: ES, NQ, YM, RTY, CL, GC

**Key Concepts:**
- **Contract Code**: NQH25 = Nasdaq (NQ) + March (H) + 2025 (25)
- **Month Codes**: F(Jan), G(Feb), H(Mar), J(Apr), K(May), M(Jun), N(Jul), Q(Aug), U(Sep), V(Oct), X(Nov), Z(Dec)
- **Rollover**: Switching from expiring contract to next active contract
- **Expiration**: Third Friday of contract month for equity index futures

@dataclass
class FuturesContract:
    id: int
    symbol_root: str        # "NQ", "ES", "CL"
    contract_code: str      # "NQH25"
    month_code: str         # "H" (March)
    year: int               # 2025

    last_trading_day: date  # Usually day before expiration
    rollover_date: date     # Suggested rollover date (e.g., 3 days before)
    is_active: bool         # Currently tradeable
    next_contract: Optional[str]  # "NQM25" (next quarter)

# Database model
class FuturesContract(Base):
    __tablename__ = "futures_contracts"

    id = Column(Integer, primary_key=True)
    symbol_root = Column(String(10), nullable=False, index=True)
    contract_code = Column(String(20), nullable=False, unique=True)
    month_code = Column(String(1), nullable=False)
    year = Column(Integer, nullable=False)
    expiration_date = Column(Date, nullable=False)
    last_trading_day = Column(Date, nullable=False)
    rollover_date = Column(Date, nullable=False)
    is_active = Column(Boolean, default=True)
    next_contract = Column(String(20))
    created_at = Column(DateTime, default=datetime.utcnow)

class FuturesContractService:

# Scheduled task (run daily)
class CheckContractExpirationsUseCase:
    async def execute(self) -> List[RolloverNotification]:
        """Check all users for expiring contracts and send notifications."""
        notifications = []

        # Get all contracts expiring in next 7 days
        expiring = await self.futures_service.get_contracts_expiring_soon(days=7)

        for contract in expiring:
            # Find users trading this contract
            users = await self.tracker.get_users_trading(contract.contract_code)

            for user in users:
                notification = RolloverNotification(
                    user_id=user.id,
                    contract_code=contract.contract_code,
                    expiration_date=contract.expiration_date,
                    suggested_rollover=contract.rollover_date,
                    next_contract=contract.next_contract,
                    days_until_expiration=(contract.expiration_date - date.today()).days
                )
                await self.notification_service.send(notification)
                notifications.append(notification)

        return notifications

# Add to notification types
class NotificationType(str, Enum):
    TRADE_EXECUTED = "trade_executed"
    CONNECTION_ERROR = "connection_error"
    CONTRACT_EXPIRING = "contract_expiring"  # New
    ROLLOVER_SUGGESTED = "rollover_suggested"  # New

// ExpirationBadge component
function ExpirationBadge({ contract }: { contract: FuturesInfo }) {
  const daysUntil = differenceInDays(contract.expirationDate, new Date());

  if (daysUntil > 7) return null;

  return (
    <Badge variant={daysUntil <= 3 ? "destructive" : "warning"}>
      <AlertTriangle className="h-3 w-3 mr-1" />
      Expires in {daysUntil} day{daysUntil !== 1 ? 's' : ''}
    </Badge>
  );
}

// Dashboard alert card
function ExpirationAlerts({ alerts }: { alerts: ExpirationAlert[] }) {
  return (
    <Card>
      <CardHeader>
        <CardTitle>Contract Expirations</CardTitle>
      </CardHeader>
      <CardContent>
        {alerts.map(alert => (
          <div key={alert.contractCode} className="flex justify-between">
            <span>{alert.contractCode}</span>
            <span>Expires {format(alert.expirationDate, 'MMM d')}</span>
            <Button size="sm" variant="outline">
              Roll to {alert.nextContract}
            </Button>
          </div>
        ))}
      </CardContent>
    </Card>
  );
}
