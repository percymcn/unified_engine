```python
class SymbolFormatDetector:
    # Known base symbols for pattern detection
    REFERENCE_SYMBOLS = [
        "EURUSD", "GBPUSD", "USDJPY", "AUDUSD",  # Forex
        "XAUUSD", "XAGUSD",  # Metals
        "US30", "US500", "NAS100", "GER40",  # Indices
        "BTCUSD", "ETHUSD",  # Crypto
    ]

    def detect_format(self, available_symbols: List[str]) -> Dict[str, Any]:
        """
        Analyze symbols to detect format patterns.
        Returns: {
            "suffix": ".pro" | ".raw" | "" | None,
            "prefix": "t" | "" | None,
            "case": "upper" | "lower" | "mixed",
            "separator": "." | "_" | "",
            "confidence": 0.0-1.0
        }
        """

    def build_symbol_map(
        self,
        available_symbols: List[str],
        detected_format: Dict[str, Any]
    ) -> Dict[str, str]:
        """
        Build mapping from standard symbols to broker format.
        Returns: {"EURUSD": "EURUSD.pro", "US30": "US30.pro", ...}
        """

    def match_symbol(
        self,
        source_symbol: str,
        available_symbols: List[str],
        detected_format: Dict[str, Any]
    ) -> Optional[str]:
        """Find best match for source symbol in available symbols."""
```

**Acceptance:**
- Detects common suffixes (.pro, .raw, .std, etc.)
- Detects case patterns (upper, lower, mixed)
- Builds mapping for common trading symbols
- Returns confidence score for detection quality

```python
# In test_connection.py
async def _test_and_detect_symbols(
    self,
    broker_type: str,
    credentials: Dict
) -> TestConnectionResult:
    """Test connection and detect symbol format."""
    # Test connection first
    result = await self._test_connection(broker_type, credentials)

    if result.success:
        # Get available symbols
        symbols = await self._get_broker_symbols(broker_type, credentials)

        # Detect format
        detector = SymbolFormatDetector()
        detected_format = detector.detect_format(symbols)
        symbol_map = detector.build_symbol_map(symbols, detected_format)

        result.detected_format = detected_format
        result.symbol_map = symbol_map
        result.sample_symbols = symbols[:20]  # First 20 for UI preview

    return result
```

**Acceptance:**
- Connection test returns detected symbol format
- Format stored when account is created
- UI can display detected format info

```python
# After account creation with detected format:
async def _create_auto_aliases(
    self,
    user_id: int,
    account_id: int,
    broker_type: str,
    symbol_map: Dict[str, str]
) -> int:
    """Create auto-detected symbol aliases."""
    created = 0
    for source, target in symbol_map.items():
        if source != target:  # Only create if different
            alias = SymbolAlias(
                user_id=user_id,
                source_symbol=source,
                broker_type=broker_type,
                target_symbol=target,
                is_auto_detected=True
            )
            await self.alias_repo.create(alias)
            created += 1
    return created
```

**Acceptance:**
- Auto-aliases created on account connection
- Marked as `is_auto_detected=True`
- User can override auto-detected aliases
