    Create immutable value objects using frozen dataclasses:

    ```python
    from dataclasses import dataclass
    from decimal import Decimal
    from typing import Optional
    from app.domain.exceptions import ValidationError

    @dataclass(frozen=True)
    class Money:
        """Represents monetary value with currency"""
        amount: Decimal
        currency: str = "USD"

        def __post_init__(self):
            if self.amount < 0:
                raise ValidationError("Money amount cannot be negative")
            if len(self.currency) != 3:
                raise ValidationError("Currency must be 3-letter code")

        def __add__(self, other: "Money") -> "Money":
            if self.currency != other.currency:
                raise ValidationError("Cannot add different currencies")
            return Money(self.amount + other.amount, self.currency)

        def __sub__(self, other: "Money") -> "Money":
            if self.currency != other.currency:
                raise ValidationError("Cannot subtract different currencies")
            return Money(self.amount - other.amount, self.currency)

    @dataclass(frozen=True)
    class Volume:
        """Trading volume/quantity"""
        value: Decimal

        def __post_init__(self):
            if self.value <= 0:
                raise ValidationError("Volume must be positive")

    @dataclass(frozen=True)
    class Price:
        """Market price"""
        value: Decimal

        def __post_init__(self):
            if self.value < 0:
                raise ValidationError("Price cannot be negative")

    @dataclass(frozen=True)
    class Symbol:
        """Trading symbol/instrument"""
        value: str

        def __post_init__(self):
            if not self.value or len(self.value) > 20:
                raise ValidationError("Invalid symbol")
            # Use object.__setattr__ for frozen dataclass
            object.__setattr__(self, 'value', self.value.upper())

    @dataclass(frozen=True)
    class AccountId:
        """Unique account identifier"""
        value: str

        def __post_init__(self):
            if not self.value:
                raise ValidationError("Account ID cannot be empty")

    @dataclass(frozen=True)
    class SignalId:
        """Unique signal identifier"""
        value: str

        def __post_init__(self):
            if not self.value:
                raise ValidationError("Signal ID cannot be empty")

    @dataclass(frozen=True)
    class OrderId:
        """Unique order identifier"""
        value: str

        def __post_init__(self):
            if not self.value:
                raise ValidationError("Order ID cannot be empty")

    @dataclass(frozen=True)
    class PositionId:
        """Unique position identifier"""
        value: str

        def __post_init__(self):
            if not self.value:
                raise ValidationError("Position ID cannot be empty")

    @dataclass(frozen=True)
    class StopLoss:
        """Stop loss price level"""
        price: Price

    @dataclass(frozen=True)
    class TakeProfit:

        price: Price
    ```
